\section[\appendixname~\thesection]{Symbol dictionary}
\label{sec:symbols}

\begin{center}
\begin{tabular}{p{0.22\linewidth} p{0.72\linewidth}}
$\vecX(x,t)\in\R^4$ & Embedding field of the brane into ambient 4D Euclidean space (Paper~I).\\
$t$ & Slicing parameter along the inside observer's timelike direction.\\
$g_{ij}$ & Induced metric: $g_{ij}=\partial_i\vecX\cdot\partial_j\vecX$.\\
$h_\star$ & Held ground-state scale imposed by boundary conditions.\\
$\alpha$ & Dimensionless pre-stress parameter: $\alpha=\ell_0/h_\star\in(0,1]$.\\
$\vecu$ & Displacement field relative to a background: $\vecu=\vecX-\vecX_\star$.\\
$u$ & Shorthand for a transverse component in a graph-type embedding.\\
$\mathbf{k}$ & Crystal momentum / wave vector.\\
$k_s$ & Central-force spring stiffness of the 6-neighbor stencil (Paper~I).\\
$m$ & Node mass (discrete lattice; continuum mass density $\rho_m=m/a^3$).\\
$a$ & Lattice spacing of the cubic substrate (microphysical length scale).\\
$c_L$ & Long-wavelength longitudinal wave speed: $c_L^2=k_s a^2/m$ (Paper~I \Cref{eq:wave-speeds-derived}).\\
$c_T$ & Long-wavelength transverse wave speed: $c_T^2=(1-\alpha)\,k_s a^2/m$ (Paper~I \Cref{eq:wave-speeds-derived}).\\
\end{tabular}
\end{center}
